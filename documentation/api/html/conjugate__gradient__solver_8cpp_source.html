80 std::vector<double> ConjugateGradientSolver::solve_from(const std::vector<std::vector<double>>& A, const std::vector<double>& b,
81 const std::vector<double>& x0, double tol, int max_iter)
82 {
83 checks::check_square_system(A, b);
84 checks::check_tolerance(tol, max_iter);
85 if (x0.size() != b.size())
86 {
87 throw std::invalid_argument("Starting guess x0 must have the same size as b.");
88 }
89 check_symmetric(A);
90
91 // Stop when ||r|| <= tol ||b||. For b = 0 the answer is x = 0, reached with ||r|| <= tol.
92 double b_norm = std::sqrt(dot(b, b));
93 double target = tol * (b_norm > 0.0 ? b_norm : 1.0);
94
95 std::vector<double> x = x0;
96 std::vector<double> Ax = multiply(A, x);
97 std::vector<double> r(b.size()); // residual b - A x
98 for (std::size_t i = 0; i < b.size(); ++i)
99 {
100 r[i] = b[i] - Ax[i];
101 }
102 std::vector<double> p = r; // first search direction: steepest descent
103 double rr = dot(r, r);
104
105 if (std::sqrt(rr) <= target)
106 {
107 return x; // the starting guess is already good enough
108 }
109
110 for (int k = 0; k < max_iter; ++k)
111 {
112 std::vector<double> Ap = multiply(A, p);
113 double pAp = dot(p, Ap);
114
115 // For an SPD matrix p.A p > 0 for every non-zero p.
116 if (!(pAp > 0.0))
117 {
118 throw std::runtime_error("Matrix is not positive definite: conjugate gradient cannot be used.");
119 }
120
121 double alpha = rr / pAp; // step length that minimises the error along p
122 for (std::size_t i = 0; i < x.size(); ++i)
123 {
124 x[i] += alpha * p[i];
125 r[i] -= alpha * Ap[i]; // updated residual, without another product A x
126 }
127
128 double rr_new = dot(r, r);
129 if (std::sqrt(rr_new) <= target)
130 {
131 return x;
132 }
133
134 // New direction: the residual made A-conjugate to the previous direction.
135 double beta = rr_new / rr;
136 for (std::size_t i = 0; i < p.size(); ++i)
137 {
138 p[i] = r[i] + beta * p[i];
139 }
140 rr = rr_new;
141 }
142
143 throw std::runtime_error("Conjugate gradient did not reach the tolerance within max_iter iterations.");
144 }
