20void accuracy()
21{
22 numcomp::NumericalDifferentiator d;
23 auto sine = [](double x) { return std::sin(x); };
24 double exact = std::cos(1.0);
25
26 check("Forward: d/dx sin at 1, h = 1e-6, close to cos(1)", close(d.forward_difference(sine, 1.0, 1e-6), exact, 1e-5));
27 check("Backward: d/dx sin at 1, h = 1e-6, close to cos(1)", close(d.backward_difference(sine, 1.0, 1e-6), exact, 1e-5));
28 check("Central: d/dx sin at 1, h = 1e-5, close to cos(1)", close(d.central_difference(sine, 1.0, 1e-5), exact, 1e-9));
29 check("Second: d2/dx2 sin at 1, h = 1e-3, close to -sin(1)", close(d.second_derivative(sine, 1.0, 1e-3), -std::sin(1.0), 1e-6));
30 check("Richardson: d/dx sin at 1, h = 0.01, close to cos(1)", close(d.richardson_extrapolation(sine, 1.0, 0.01), exact, 1e-10));
31 check("Central: d/dx e^x at 0 is 1", close(d.central_difference([](double x) { return std::exp(x); }, 0.0, 1e-5), 1.0, 1e-9));
32
33 // Polynomials: the central formula is exact for quadratics, the second-derivative formula for cubics.
34 check("Central: exact for x^2 at x = 3 (gives 6)", close(d.central_difference([](double x) { return x * x; }, 3.0, 0.5), 6.0, 1e-12));
35 check("Second: exact for x^3 at x = 2 (gives 12)", close(d.second_derivative([](double x) { return x * x * x; }, 2.0, 0.1), 12.0, 1e-9));
36 check("Forward: exact for a straight line 3x + 1", close(d.forward_difference([](double x) { return 3.0 * x + 1.0; }, 5.0, 0.25), 3.0, 1e-12));
37
38 // Order of accuracy: halving h divides the error by about 2, 4 and 16.
39 double r_fwd = std::abs(d.forward_difference(sine, 1.0, 0.1) - exact) / std::abs(d.forward_difference(sine, 1.0, 0.05) - exact);
40 double r_cen = std::abs(d.central_difference(sine, 1.0, 0.1) - exact) / std::abs(d.central_difference(sine, 1.0, 0.05) - exact);
41 double r_sec = std::abs(d.second_derivative(sine, 1.0, 0.1) + std::sin(1.0)) / std::abs(d.second_derivative(sine, 1.0, 0.05) + std::sin(1.0));
42 double r_ric = std::abs(d.richardson_extrapolation(sine, 1.0, 0.1) - exact) / std::abs(d.richardson_extrapolation(sine, 1.0, 0.05) - exact);
43 check("Forward: halving h halves the error (first order)", r_fwd > 1.8 && r_fwd < 2.2);
44 check("Central: halving h divides the error by about 4", r_cen > 3.8 && r_cen < 4.2);
45 check("Second: halving h divides the error by about 4", r_sec > 3.8 && r_sec < 4.2);
46 check("Richardson: halving h divides the error by about 16", r_ric > 14.0 && r_ric < 18.0);
47}

50void edge_cases()
51{
52 numcomp::NumericalDifferentiator d;
53 check("Central: works at a negative x (d/dx x^2 at -2 is -4)", close(d.central_difference([](double x) { return x * x; }, -2.0, 1e-3), -4.0, 1e-9));
54 check("Second: constant function gives 0", close(d.second_derivative([](double) { return 7.0; }, 0.0, 0.1), 0.0, 1e-12));
55 // |x| has no derivative at 0: the one-sided formulas see the two different slopes.
56 auto abs_f = [](double x) { return std::abs(x); };
57 check("Forward at the corner of |x| gives the right slope +1", close(d.forward_difference(abs_f, 0.0, 0.1), 1.0, 1e-12));
58 check("Backward at the corner of |x| gives the left slope -1", close(d.backward_difference(abs_f, 0.0, 0.1), -1.0, 1e-12));
59 check("Central at the corner of |x| gives the average 0", close(d.central_difference(abs_f, 0.0, 0.1), 0.0, 1e-12));
60}
