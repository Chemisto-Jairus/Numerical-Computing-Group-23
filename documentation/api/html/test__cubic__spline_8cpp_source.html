20void normal_cases()

24 std::vector<double> x = {0.0, 1.0, 2.0, 3.0, 4.0};

26 bool through_points = true;
27 for (std::size_t i = 0; i < x.size(); ++i)
28 {
29 if (!close(spline.interpolate(x[i], x, y), y[i], 1e-12)) through_points = false;
30 }
31 check("Spline goes exactly through every data point", through_points);
32
33 // Points (0,0), (1,1), (2,0). By hand: 4*M1 = -12, so M1 = -3.
34 // S(0.5) = -3*(0.5^3)/6 + (1 - (-3)/6)*0.5 = -0.0625 + 0.75 = 0.6875
35 std::vector<double> hx = {0.0, 1.0, 2.0};
36 std::vector<double> hy = {0.0, 1.0, 0.0};
37 check("Hand-worked value S(0.5) = 0.6875", close(spline.interpolate(0.5, hx, hy), 0.6875, 1e-12));
38 check("Symmetry: S(1.5) = S(0.5)", close(spline.interpolate(1.5, hx, hy), 0.6875, 1e-12));
39
40 std::vector<double> M = spline.second_derivatives(hx, hy);
41 check("second_derivatives gives {0, -3, 0}", M.size() == 3 && close(M[0], 0.0, 1e-12) && close(M[1], -3.0, 1e-12) && close(M[2], 0.0, 1e-12));
42
43 std::vector<double> M2 = spline.second_derivatives({0.0, 0.5, 1.5, 2.0, 3.5}, {2.0, -1.0, 4.0, 0.0, 1.0});
44 check("Natural spline: M is zero at both ends", M2.front() == 0.0 && M2.back() == 0.0);
45
46 check("Straight line data: S(2.5) = 6", close(spline.interpolate(2.5, {0.0, 1.0, 2.0, 3.0}, {1.0, 3.0, 5.0, 7.0}), 6.0, 1e-12));
47
48 // 11 points of sin(x) on [0, pi]: the spline should be very close to sin
49 const double pi = std::acos(-1.0);
50 std::vector<double> sx, sy;
51 for (int i = 0; i <= 10; ++i)
52 {
53 sx.push_back(i * pi / 10.0);
54 sy.push_back(std::sin(i * pi / 10.0));
55 }

58 {
59 double xt = (j + 0.5) * pi / 20.0;
60 if (!close(spline.interpolate(xt, sx, sy), std::sin(xt), 1e-3)) near_sin = false;
61 }
62 check("Spline of sin(x) with 11 points is within 1e-3", near_sin);
