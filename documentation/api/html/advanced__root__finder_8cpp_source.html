14 double AdvancedRootFinder::secant_method(std::function<double(double)> f, double x0, double x1, double tol, int max_iter)
15 {
16 checks::check_tolerance(tol, max_iter);
17
18 for (int iter = 0; iter < max_iter; ++iter)
19 {
20 double f_x0 = f(x0);
21 double f_x1 = f(x1);
22
23 // Equal function values give a horizontal secant line with no zero crossing.
24 if (std::abs(f_x1 - f_x0) < 1e-12)
25 {
26 throw std::runtime_error("Division by zero in Secant Method.");
27 }
28
29 // Zero crossing of the line through (x0, f(x0)) and (x1, f(x1)).
30 double x2 = x1 - f_x1 * (x1 - x0) / (f_x1 - f_x0);
31 if (std::abs(x2 - x1) < tol) return x2;
32
33 // The two most recent points define the next secant line.
34 x0 = x1;
35 x1 = x2;
36 }
37 throw std::runtime_error("Secant method failed to converge.");
38 }

40 double AdvancedRootFinder::fixed_point_iteration(std::function<double(double)> g, double x0, double tol, int max_iter)
41 {
42 checks::check_tolerance(tol, max_iter);
43
44 double x_current = x0;
45 for (int iter = 0; iter < max_iter; ++iter)
46 {
47 double x_next = g(x_current);
48
49 // Successive values agree to within tol, so x_next is (nearly) a fixed point.
50 if (std::abs(x_next - x_current) < tol) return x_next;
51 x_current = x_next;
52 }
