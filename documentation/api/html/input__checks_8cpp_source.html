58 std::size_t find_interval(double x, const std::vector<double>& x_vals)
59 {
60 // Written as !(...) so that a NaN x is rejected as well.
61 if (!(x >= x_vals.front() && x <= x_vals.back()))
62 {
63 throw std::out_of_range("Target x is out of the interpolation bounds.");
64 }
65
66 // Binary search: upper_bound returns the first x value that is greater than x.
67 // The interval starts one position before it. x equal to the last data point
68 // belongs to the last interval.
69 std::size_t i = std::upper_bound(x_vals.begin(), x_vals.end(), x) - x_vals.begin();
70 if (i >= x_vals.size())
71 {
72 i = x_vals.size() - 1;
73 }
74 return i - 1;
75 }
