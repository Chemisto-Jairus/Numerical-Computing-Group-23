53void chebyshev()
54{
55 numcomp::Interpolator interp;
56
57 std::vector<double> two = interp.chebyshev_nodes(-1.0, 1.0, 2);
58 double r = std::sqrt(2.0) / 2.0; // cos(pi/4)
59 check("Chebyshev: n = 2 on [-1,1] gives +/- 0.7071", two.size() == 2 && close(two[0], r, 1e-12) && close(two[1], -r, 1e-12));
60
61 std::vector<double> seven = interp.chebyshev_nodes(2.0, 5.0, 7);
62 bool inside = seven.size() == 7;
63 for (double node : seven)
64 {
65 if (node <= 2.0 || node >= 5.0) inside = false;
66 }
67 check("Chebyshev: all 7 nodes lie strictly inside [2, 5]", inside);
68
69 std::vector<double> one = interp.chebyshev_nodes(0.0, 4.0, 1);
70 check("Chebyshev: n = 1 gives the midpoint", one.size() == 1 && close(one[0], 2.0, 1e-12));
71
72 test::expect_throw<std::invalid_argument>("Chebyshev: n = 0 throws invalid_argument", [&] { interp.chebyshev_nodes(0.0, 1.0, 0); });
73 test::expect_throw<std::invalid_argument>("Chebyshev: n < 0 throws invalid_argument", [&] { interp.chebyshev_nodes(0.0, 1.0, -3); });
74 test::expect_throw<std::invalid_argument>("Chebyshev: a == b throws invalid_argument", [&] { interp.chebyshev_nodes(1.0, 1.0, 3); });
75 test::expect_throw<std::invalid_argument>("Chebyshev: a > b throws invalid_argument", [&] { interp.chebyshev_nodes(2.0, 1.0, 3); });
76 // The node count is capped so the allocation stays bounded.
77 test::expect_throw<std::invalid_argument>("Chebyshev: n > 1,000,000 throws invalid_argument", [&] { interp.chebyshev_nodes(0.0, 1.0, 1000000001); });
78 test::expect_throw<std::invalid_argument>("Chebyshev: NaN end point throws invalid_argument", [&] { interp.chebyshev_nodes(std::nan(""), 1.0, 3); });
79}

82void chebyshev_approximation()
83{
84 numcomp::Interpolator interp;
85 auto exp_f = [](double x) { return std::exp(x); };
86
87 // e^x is smooth, so 12 coefficients reach almost full double precision.
88 std::vector<double> c = interp.chebyshev_coefficients(exp_f, 0.0, 1.0, 12);
89 double worst = 0.0;
90 for (int k = 0; k <= 20; ++k)
91 {
92 double x = k / 20.0;
93 worst = std::max(worst, std::abs(interp.chebyshev_approximate(c, 0.0, 1.0, x) - std::exp(x)));
94 }
95 check("Chebyshev approx: e^x on [0, 1] with 12 terms, error < 1e-13", worst < 1e-13);
96 check("Chebyshev approx: coefficients of e^x shrink quickly", std::abs(c[11]) < 1e-12 && std::abs(c[1]) > 0.1);
97
98 // A polynomial of degree 3 is reproduced exactly with 4 terms.
99 auto cubic = [](double x) { return 2.0 * x * x * x - x + 1.0; };
100 std::vector<double> cc = interp.chebyshev_coefficients(cubic, -2.0, 3.0, 4);
101 check("Chebyshev approx: cubic reproduced exactly at x = 0.7", close(interp.chebyshev_approximate(cc, -2.0, 3.0, 0.7), cubic(0.7), 1e-11));
102
103 // Edge cases: interval end points, and a single coefficient (a constant).
104 check("Chebyshev approx: exact at the left end a", close(interp.chebyshev_approximate(c, 0.0, 1.0, 0.0), 1.0, 1e-13));
105 check("Chebyshev approx: exact at the right end b", close(interp.chebyshev_approximate(c, 0.0, 1.0, 1.0), std::exp(1.0), 1e-13));
106 std::vector<double> one = interp.chebyshev_coefficients([](double) { return 5.0; }, 0.0, 1.0, 1);
107 check("Chebyshev approx: n = 1 gives the constant 5", close(interp.chebyshev_approximate(one, 0.0, 1.0, 0.3), 5.0, 1e-13));
108
109 // More terms give a smaller error for the Runge function (no oscillation at Chebyshev nodes).
110 auto runge = [](double x) { return 1.0 / (1.0 + 25.0 * x * x); };
111 double err10 = 0.0, err40 = 0.0;
112 std::vector<double> r10 = interp.chebyshev_coefficients(runge, -1.0, 1.0, 10);
113 std::vector<double> r40 = interp.chebyshev_coefficients(runge, -1.0, 1.0, 40);
114 for (int k = 0; k <= 100; ++k)
115 {
116 double x = -1.0 + k * 0.02;
117 err10 = std::max(err10, std::abs(interp.chebyshev_approximate(r10, -1.0, 1.0, x) - runge(x)));
118 err40 = std::max(err40, std::abs(interp.chebyshev_approximate(r40, -1.0, 1.0, x) - runge(x)));
119 }
120 check("Chebyshev approx: Runge function error falls from 10 to 40 terms", err40 < err10 / 10.0);
121
122 // Invalid input
123 test::expect_throw<std::invalid_argument>("Chebyshev approx: n = 0 throws invalid_argument", [&] { interp.chebyshev_coefficients(exp_f, 0.0, 1.0, 0); });
124 test::expect_throw<std::invalid_argument>("Chebyshev approx: n > 1000 throws invalid_argument", [&] { interp.chebyshev_coefficients(exp_f, 0.0, 1.0, 1001); });
125 test::expect_throw<std::invalid_argument>("Chebyshev approx: a > b throws invalid_argument", [&] { interp.chebyshev_coefficients(exp_f, 1.0, 0.0, 5); });
126 test::expect_throw<std::runtime_error>("Chebyshev approx: f returning NaN throws runtime_error", [&] { interp.chebyshev_coefficients([](double x) { return std::log(x); }, -1.0, 1.0, 4); });
127 test::expect_throw<std::invalid_argument>("Chebyshev approx: empty coefficients throw invalid_argument", [&] { interp.chebyshev_approximate({}, 0.0, 1.0, 0.5); });
128 test::expect_throw<std::out_of_range>("Chebyshev approx: x outside [a, b] throws out_of_range", [&] { interp.chebyshev_approximate(c, 0.0, 1.0, 1.5); });
129 test::expect_throw<std::out_of_range>("Chebyshev approx: NaN x throws out_of_range", [&] { interp.chebyshev_approximate(c, 0.0, 1.0, std::nan("")); });
130}
