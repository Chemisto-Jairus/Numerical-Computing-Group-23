46 check("AB4: decay dy/dt = -2y, h = 0.05, close to e^-2", close(ab.ab4(decay_rhs, 0.0, 1.0, 1.0, 0.05).back().second, exact, 1e-5));
47 check("AB2: decay dy/dt = -2y, h = 0.05, close to e^-2", close(ab.ab2(decay_rhs, 0.0, 1.0, 1.0, 0.05).back().second, exact, 2e-3));
48 // When f depends only on t, ABk is exact for slopes that are polynomials of degree k - 1.
49 check("AB2: dy/dt = 2t gives y(2) = 4 exactly", close(ab.ab2(linear_rhs, 0.0, 0.0, 2.0, 0.1).back().second, 4.0, 1e-12));
50 check("AB4: dy/dt = 3t^2 gives y(2) = 8 exactly", close(ab.ab4(square_rhs, 0.0, 0.0, 2.0, 0.1).back().second, 8.0, 1e-10));

70 // Fewer steps than the method needs: only the RK4 starter runs, and the answer is still accurate.
