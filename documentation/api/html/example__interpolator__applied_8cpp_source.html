28template <typename F>
29double max_linear_error(numcomp::Interpolator& interp, F f, const std::vector<double>& xs, double a, double b)
30{
31 std::vector<double> ys;
32 for (double x : xs) ys.push_back(f(x));
33 double worst = 0.0;
34 for (int k = 0; k <= 2000; ++k)
35 {
36 double t = a + (b - a) * k / 2000.0;
37 worst = std::max(worst, std::abs(interp.linear_interpolate(t, xs, ys) - f(t)));
38 }
39 return worst;
40}
