20 double Interpolator::linear_interpolate(double x, const std::vector<double>& x_vals, const std::vector<double>& y_vals)
21 {
22 checks::check_data_points(x_vals, y_vals);
23 std::size_t i = checks::find_interval(x, x_vals);
24
25 double x0 = x_vals[i], x1 = x_vals[i+1];
26 double y0 = y_vals[i], y1 = y_vals[i+1];
27
28 // Point on the straight line through (x0, y0) and (x1, y1).
29 return y0 + (y1 - y0) * (x - x0) / (x1 - x0);

32 std::vector<double> Interpolator::chebyshev_nodes(double a, double b, int n)
33 {
34 if (n <= 0)
35 {
36 throw std::invalid_argument("Number of nodes must be strictly positive.");
37 }
38
39 // Shared memory limit with the ODE solvers.
40 if (n > checks::MAX_POINTS)
41 {
42 throw std::invalid_argument("Number of nodes must not be more than 1,000,000.");
43 }
44
45 // Written as !(a < b) so that NaN end points are rejected as well.
46 if (!(a < b))
47 {
48 throw std::invalid_argument("a must be less than b.");
49 }
50
51 std::vector<double> nodes(n);
52 for (int k = 1; k <= n; ++k)
53 {
54 // Root of T_n on [-1, 1], shifted and scaled onto [a, b].
55 nodes[k-1] = 0.5 * (a + b) + 0.5 * (b - a) * std::cos((2.0 * k - 1.0) * M_PI / (2.0 * n));
56 }
57 return nodes;
58 }

63 std::vector<double> Interpolator::chebyshev_coefficients(std::function<double(double)> f, double a, double b, int n)
64 {
65 if (n > MAX_CHEBYSHEV_TERMS)
66 {
67 throw std::invalid_argument("Number of Chebyshev coefficients must not be more than 1,000.");
68 }
69
70 // Reuses the node formula (and its checks of a, b and n) instead of repeating it.
71 std::vector<double> nodes = chebyshev_nodes(a, b, n);
72
73 // Sample f once at every node.
74 std::vector<double> fx(n);
75 for (int k = 0; k < n; ++k)
76 {
77 fx[k] = f(nodes[k]);
78 checks::check_finite_result(fx[k]);
79 }
80
81 // c_j = (2/n) sum_k f(x_k) cos(j theta_k), with theta_k = (2k + 1) pi / (2n) for k = 0..n-1.
82 std::vector<double> coeffs(n, 0.0);
83 for (int j = 0; j < n; ++j)
84 {
85 double sum = 0.0;
86 for (int k = 0; k < n; ++k)
87 {
88 sum += fx[k] * std::cos(j * (2.0 * k + 1.0) * M_PI / (2.0 * n));
89 }
90 coeffs[j] = 2.0 * sum / n;
91 }
92 return coeffs;
93 }

95 double Interpolator::chebyshev_approximate(const std::vector<double>& coeffs, double a, double b, double x)
96 {
97 if (coeffs.empty())
98 {
99 throw std::invalid_argument("The list of Chebyshev coefficients must not be empty.");
100 }
101 if (!(a < b))
102 {
103 throw std::invalid_argument("a must be less than b.");
104 }
105 // Written as !(...) so that a NaN x is rejected as well.
106 if (!(x >= a && x <= b))
107 {
108 throw std::out_of_range("x is outside the interval [a, b] of the approximation.");
109 }
110
111 // Map [a, b] onto [-1, 1], where the Chebyshev polynomials are defined.
112 double u = (2.0 * x - a - b) / (b - a);
113
114 // Clenshaw's recurrence, from the highest coefficient down to c_1.
115 double b1 = 0.0; // b_(j+1)
116 double b2 = 0.0; // b_(j+2)
117 for (std::size_t j = coeffs.size() - 1; j >= 1; --j)
118 {
119 double bj = 2.0 * u * b1 - b2 + coeffs[j];
120 b2 = b1;
121 b1 = bj;
122 }
123 return u * b1 - b2 + 0.5 * coeffs[0];
124 }
