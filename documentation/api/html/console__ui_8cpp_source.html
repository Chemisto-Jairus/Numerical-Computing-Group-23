125 double read_double(const std::string& prompt, double default_value)
126 {
127 std::ostringstream shown;
128 shown << prompt << " [" << default_value << "]: ";
129 while (true)
130 {
131 std::string text = trim(read_line(shown.str()));
132 if (text.empty())
133 {
134 return default_value;
135 }
136 try
137 {
138 std::size_t used = 0;
139 double value = std::stod(text, &used);
140 if (used == text.size() && std::isfinite(value))
141 {
142 return value;
143 }
144 }
145 catch (const std::exception&)
146 {
147 // Not a number: fall through and ask again.
148 }
149 print_error("Invalid input", "please type a number such as 2, -0.5 or 1e-6.");
150 }
151 }

153 int read_int(const std::string& prompt, int default_value, int min_value, int max_value)
154 {
155 // A default outside the range means "no default": an empty line is rejected.
156 bool has_default = default_value >= min_value && default_value <= max_value;
157 std::ostringstream shown;
158 shown << prompt;
159 if (has_default)
160 {
161 shown << "[" << default_value << "]: ";
162 }
163 while (true)
164 {
165 std::string text = trim(read_line(shown.str()));
166 if (text.empty() && has_default)
167 {
168 return default_value;
169 }
170 try
171 {
172 std::size_t used = 0;
173 long value = std::stol(text, &used);
174 if (used == text.size() && value >= min_value && value <= max_value)
175 {
176 return static_cast<int>(value);
177 }
178 }
179 catch (const std::exception&)
180 {
181 // Not a whole number: fall through and ask again.
182 }
183 print_error("Invalid input", "please type a whole number from " + std::to_string(min_value) + " to " + std::to_string(max_value) + ".");
184 }

188 {
189 while (true)
190 {
191 std::string text = trim(read_line(prompt + (default_value ? " [Y/n]: " : " [y/N]: ")));
192 if (text.empty()) return default_value;
193 if (text == "y" || text == "Y" || text == "yes") return true;
194 if (text == "n" || text == "N" || text == "no") return false;

246 void print_error(const std::string& kind, const std::string& message)
247 {
248 // Long messages are wrapped at word boundaries so no line exceeds WIDTH;
249 // continuation lines are indented under the start of the message.
250 std::string line = " [!] " + kind + ": ";
251 const std::string indent = " ";
252 std::istringstream words(message);
253 std::string word;
254 bool line_has_word = false;
255 while (words >> word)
256 {
257 if (line_has_word && line.size() + 1 + word.size() > static_cast<std::size_t>(WIDTH))
258 {
259 std::cout << line << "\n";
260 line = indent;
261 line_has_word = false;
262 }
263 if (line_has_word) line += " ";
264 line += word;
265 line_has_word = true;
266 }
267 std::cout << line << "\n";
268 }
