66
67 // x0 == x1 means f(x1) - f(x0) = 0, so the method would divide by zero

69 // g(x) = 2x + 1 runs away from its fixed point (-1), so it never converges
70 test::expect_throw<std::runtime_error>("Fixed-point: diverging g throws runtime_error", [&] { finder.fixed_point_iteration([](double x) { return 2.0 * x + 1.0; }, 1.0, 1e-8, 100); });
71}
