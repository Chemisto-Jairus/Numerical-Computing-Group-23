26 static int find_pivot_row(const std::vector<std::vector<double>>& M, int k)
27 {
28 int n = static_cast<int>(M.size());
29 int pivot_row = k;
30 for (int i = k + 1; i < n; ++i)
31 {
32 if (std::abs(M[i][k]) > std::abs(M[pivot_row][k]))
33 {
34 pivot_row = i;
35 }
36 }
37

55 static std::vector<double> back_substitution(const std::vector<std::vector<double>>& U, const std::vector<double>& y)
56 {
57 int n = static_cast<int>(U.size());
58 std::vector<double> x(n, 0.0);
59 for (int i = n - 1; i >= 0; --i)
60 {
61 double sum = 0.0;
62 for (int j = i + 1; j < n; ++j)
63 {
64 sum += U[i][j] * x[j];
65 }
66 x[i] = (y[i] - sum) / U[i][i];
67 }

71 std::vector<double> MatrixSolver::gaussian_elimination(const std::vector<std::vector<double>>& A, const std::vector<double>& b)
72 {
73 checks::check_square_system(A, b);
74
75 int n = static_cast<int>(A.size());
76
77 // Copies: the elimination overwrites them, the caller's data stays intact.
78 std::vector<std::vector<double>> M = A;
79 std::vector<double> rhs = b;
80
81 // Forward elimination, column by column.
82 for (int k = 0; k < n; ++k)
83 {
84 int p = find_pivot_row(M, k);
85 if (p != k)
86 {
87 std::swap(M[k], M[p]);
88 std::swap(rhs[k], rhs[p]);
89 }
90
91 // Subtract a multiple of the pivot row to zero every entry below the pivot.
92 for (int i = k + 1; i < n; ++i)
93 {
94 double factor = M[i][k] / M[k][k];
95 for (int j = k; j < n; ++j)
96 {
97 M[i][j] = M[i][j] - factor * M[k][j];
98 }
99 rhs[i] = rhs[i] - factor * rhs[k];
100 }
101 }
102
103 // M is now upper triangular.
104 return back_substitution(M, rhs);
105 }

107 std::vector<double> MatrixSolver::lu_solve(const std::vector<std::vector<double>>& A, const std::vector<double>& b)
108 {
109 checks::check_square_system(A, b);
110
111 int n = static_cast<int>(A.size());
112
113 // U starts as A and is reduced to upper triangular form.
114 // L starts as the identity and collects the elimination multipliers.
115 // rhs receives the same row swaps as A (that is, it becomes P b).
116 std::vector<std::vector<double>> U = A;
117 std::vector<std::vector<double>> L(n, std::vector<double>(n, 0.0));
118 for (int i = 0; i < n; ++i)
119 {
120 L[i][i] = 1.0;
121 }
122 std::vector<double> rhs = b;
123
124 // Factorisation P A = L U.
125 for (int k = 0; k < n; ++k)
126 {
127 int p = find_pivot_row(U, k);
128 if (p != k)
129 {
130 std::swap(U[k], U[p]);
131 std::swap(rhs[k], rhs[p]);
132
133 // Only the multipliers already stored (columns 0..k-1) move with the row;
134 // the unit diagonal of L stays in place.
135 for (int j = 0; j < k; ++j)
136 {
137 std::swap(L[k][j], L[p][j]);
138 }
139 }
140
141 for (int i = k + 1; i < n; ++i)
142 {
143 L[i][k] = U[i][k] / U[k][k];
144 for (int j = k; j < n; ++j)
145 {
146 U[i][j] = U[i][j] - L[i][k] * U[k][j];
147 }
148 }
149 }
150
151 // Forward substitution L y = P b. L has a unit diagonal, so no division is needed.
152 std::vector<double> y(n, 0.0);
153 for (int i = 0; i < n; ++i)
154 {
155 double sum = 0.0;
156 for (int j = 0; j < i; ++j)
157 {
158 sum += L[i][j] * y[j];
159 }
160 y[i] = rhs[i] - sum;
161 }
162
163 // Back substitution U x = y.
164 return back_substitution(U, y);
165 }
166}
