36void accuracy()
37{
38 numcomp::AdvancedODESolver solver;
39 const double pi = std::acos(-1.0);
40 double exact = std::exp(-2.0);

42 // Polynomial: when f depends only on t, RK3 and RK4 act like Simpson's rule (exact for cubics)
43 check("RK4: dy/dt = 4t^3 gives y(2) = 16 exactly", close(solver.rk4(cubic_rhs, 0.0, 0.0, 2.0, 0.1).back().second, 16.0, 1e-10));
44 check("RK3: dy/dt = 4t^3 gives y(2) = 16 exactly", close(solver.rk3(cubic_rhs, 0.0, 0.0, 2.0, 0.1).back().second, 16.0, 1e-10));
45
46 // Trig
47 check("RK4: dy/dt = -sin(t) gives cos(pi) = -1", close(solver.rk4(sin_rhs, 0.0, 1.0, pi, 0.01).back().second, -1.0, 1e-9));
48 check("RK3: dy/dt = -sin(t) gives cos(pi) = -1", close(solver.rk3(sin_rhs, 0.0, 1.0, pi, 0.01).back().second, -1.0, 1e-8));
49
50 // Decay
51 check("RK4: decay dy/dt = -2y, h = 0.1, close to e^-2", close(solver.rk4(decay_rhs, 0.0, 1.0, 1.0, 0.1).back().second, exact, 1e-5));
52 check("RK3: decay dy/dt = -2y, h = 0.1, close to e^-2", close(solver.rk3(decay_rhs, 0.0, 1.0, 1.0, 0.1).back().second, exact, 1e-3));
53
54 // Order of accuracy: halving h divides RK3's error by about 8 and RK4's by about 16
55 double rk3_ratio = std::abs(solver.rk3(decay_rhs, 0.0, 1.0, 1.0, 0.1).back().second - exact)
56 / std::abs(solver.rk3(decay_rhs, 0.0, 1.0, 1.0, 0.05).back().second - exact);
57 double rk4_ratio = std::abs(solver.rk4(decay_rhs, 0.0, 1.0, 1.0, 0.1).back().second - exact)
58 / std::abs(solver.rk4(decay_rhs, 0.0, 1.0, 1.0, 0.05).back().second - exact);
59 check("RK3: halving h divides the error by about 8", rk3_ratio > 7.0 && rk3_ratio < 9.0);
60 check("RK4: halving h divides the error by about 16", rk4_ratio > 14.0 && rk4_ratio < 18.0);
61}
