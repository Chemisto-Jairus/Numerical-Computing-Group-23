43void exact_results()
44{
45 numcomp::AdvancedIntegrator integrator;
46 // Simpson is exact for cubics, even with only 2 strips
47 check("Simpson: x^3 on [0, 2] with n = 2 is exactly 4", close(integrator.simpsons_rule(cube, 0.0, 2.0, 2), 4.0, 1e-12));

54void edge_cases()
55{
56 numcomp::AdvancedIntegrator integrator;
