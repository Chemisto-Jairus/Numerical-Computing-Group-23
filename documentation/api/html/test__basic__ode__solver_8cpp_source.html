39void accuracy()
40{
41 numcomp::BasicODESolver solver;
42 double e = std::exp(1.0);
43
44 // Polynomial
45 check("Euler: dy/dt = 1 gives y(1) = 1 exactly", close(solver.euler(constant_rhs, 0.0, 0.0, 1.0, 0.1).back().second, 1.0, 1e-12));
46 // Euler with h = 0.1 gives exactly h^2 * n(n-1) = 0.9 (a known first-order error)
47 check("Euler: dy/dt = 2t with h = 0.1 gives 0.9 at t = 1", close(solver.euler(linear_rhs, 0.0, 0.0, 1.0, 0.1).back().second, 0.9, 1e-12));
48 // Heun averages the slopes, so it is exact when f is linear in t
49 check("Heun: dy/dt = 2t gives y(1) = 1 exactly", close(solver.heun(linear_rhs, 0.0, 0.0, 1.0, 0.1).back().second, 1.0, 1e-12));
50
51 // Trig
52 check("Euler: dy/dt = cos(t), h = 0.001, close to sin(2)", close(solver.euler(cos_rhs, 0.0, 0.0, 2.0, 0.001).back().second, std::sin(2.0), 1e-3));
53 check("Heun: dy/dt = cos(t), h = 0.01, close to sin(2)", close(solver.heun(cos_rhs, 0.0, 0.0, 2.0, 0.01).back().second, std::sin(2.0), 1e-4));
54
55 // Exponential
56 check("Euler: dy/dt = y, h = 0.001, close to e", close(solver.euler(growth_rhs, 0.0, 1.0, 1.0, 0.001).back().second, e, 2e-3));
57 check("Heun: dy/dt = y, h = 0.01, close to e", close(solver.heun(growth_rhs, 0.0, 1.0, 1.0, 0.01).back().second, e, 1e-4));
58
59 // Order of accuracy: halving h halves Euler's error and quarters Heun's
60 double euler_ratio = std::abs(solver.euler(growth_rhs, 0.0, 1.0, 1.0, 0.01).back().second - e)
61 / std::abs(solver.euler(growth_rhs, 0.0, 1.0, 1.0, 0.005).back().second - e);
62 double heun_ratio = std::abs(solver.heun(growth_rhs, 0.0, 1.0, 1.0, 0.01).back().second - e)
63 / std::abs(solver.heun(growth_rhs, 0.0, 1.0, 1.0, 0.005).back().second - e);
64 check("Euler: halving h halves the error (ratio close to 2)", euler_ratio > 1.8 && euler_ratio < 2.2);
65 check("Heun: halving h quarters the error (ratio close to 4)", heun_ratio > 3.6 && heun_ratio < 4.4);
66}
