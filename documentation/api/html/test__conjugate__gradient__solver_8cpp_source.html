45Matrix second_difference_matrix(int n)
46{
47 Matrix A(n, std::vector<double>(n, 0.0));
48 for (int i = 0; i < n; ++i)
49 {
50 A[i][i] = 2.0;
51 if (i > 0) A[i][i-1] = -1.0;
52 if (i < n - 1) A[i][i+1] = -1.0;
53 }
54 return A;
55}

58void normal_cases()
59{
60 numcomp::ConjugateGradientSolver cg;
61
62 check("2 x 2: [[4,1],[1,3]] x = [1,2] gives (1/11, 7/11)",
63 vectors_close(cg.solve({{4, 1}, {1, 3}}, {1, 2}, 1e-12, 100), {1.0 / 11.0, 7.0 / 11.0}, 1e-10));
64 check("3 x 3 SPD system gives (1, 2, 3)",
65 vectors_close(cg.solve({{4, 1, 0}, {1, 3, 1}, {0, 1, 2}}, {6, 10, 8}, 1e-12, 100), {1, 2, 3}, 1e-10));
66 check("Diagonal matrix: x_i = b_i / A_ii",
67 vectors_close(cg.solve({{2, 0, 0}, {0, 4, 0}, {0, 0, 8}}, {2, 2, 2}, 1e-12, 10), {1, 0.5, 0.25}, 1e-12));
68
69 // The 50 x 50 second-difference system with b = (1, 0, ..., 0, 1) has the solution x = (1, ..., 1).
70 // In exact arithmetic CG needs at most n iterations, so max_iter = 50 must be enough.
71 int n = 50;
72 std::vector<double> b(n, 0.0);
73 b[0] = 1.0;
74 b[n-1] = 1.0;
75 std::vector<double> x = cg.solve(second_difference_matrix(n), b, 1e-10, n);
76 check("50 x 50 tridiagonal SPD system solved within n iterations", vectors_close(x, std::vector<double>(n, 1.0), 1e-7));
77
78 // A good starting guess is returned unchanged.
79 check("solve_from: the exact answer as x0 is returned immediately",
80 vectors_close(cg.solve_from({{4, 1}, {1, 3}}, {1, 2}, {1.0 / 11.0, 7.0 / 11.0}, 1e-12, 1), {1.0 / 11.0, 7.0 / 11.0}, 1e-14));
81 check("solve_from: a rough guess still converges",
82 vectors_close(cg.solve_from({{4, 1}, {1, 3}}, {1, 2}, {10, -10}, 1e-12, 100), {1.0 / 11.0, 7.0 / 11.0}, 1e-10));
83}
