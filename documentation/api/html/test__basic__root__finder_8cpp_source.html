46void normal_cases()
47{
48 numcomp::BasicRootFinder finder;
49 check("Bisection: x^2 - 4 on [0, 3] gives 2", close(finder.bisection(f_square, 0.0, 3.0, 1e-8, 100), 2.0, 1e-6));
50 check("Bisection: x^3 - x - 2 on [1, 2]", close(finder.bisection(f_cubic, 1.0, 2.0, 1e-10, 200), 1.5213797068, 1e-7));
51 check("Newton: x^2 - 4 from x0 = 3 gives 2", close(finder.newton_raphson(f_square, df_square, 3.0, 1e-10, 50), 2.0, 1e-9));
52 check("Newton: x^3 - x - 2 from x0 = 1.5", close(finder.newton_raphson(f_cubic, df_cubic, 1.5, 1e-12, 50), 1.5213797068, 1e-9));
53 check("Newton: cos(x) - x from x0 = 1", close(finder.newton_raphson(f_cos, df_cos, 1.0, 1e-12, 50), 0.7390851332, 1e-9));

57void edge_cases()
58{
59 numcomp::BasicRootFinder finder;
60 // A reversed interval must give the same root as the ordered one.
61 check("Bisection: reversed interval [3, 0] still gives 2", close(finder.bisection(f_square, 3.0, 0.0, 1e-8, 100), 2.0, 1e-6));
62 check("Bisection: [-3, 0] finds the negative root -2", close(finder.bisection(f_square, -3.0, 0.0, 1e-8, 100), -2.0, 1e-6));
63 check("Bisection: root exactly at a = 2 is returned", finder.bisection(f_square, 2.0, 5.0, 1e-8, 100) == 2.0);
64 check("Bisection: root exactly at b = 2 is returned", finder.bisection(f_square, 0.0, 2.0, 1e-8, 100) == 2.0);
65 check("Newton: starting on the root stays there", close(finder.newton_raphson(f_square, df_square, 2.0, 1e-10, 5), 2.0, 1e-12));
