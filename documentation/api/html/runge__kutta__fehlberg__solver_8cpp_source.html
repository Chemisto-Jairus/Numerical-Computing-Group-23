32 static FehlbergStep fehlberg_step(const std::function<double(double, double)>& f, double t, double y, double h)
33 {
34 double k1 = h * f(t, y);
35 double k2 = h * f(t + h / 4.0, y + k1 / 4.0);
36 double k3 = h * f(t + 3.0 * h / 8.0, y + 3.0 * k1 / 32.0 + 9.0 * k2 / 32.0);
37 double k4 = h * f(t + 12.0 * h / 13.0, y + 1932.0 * k1 / 2197.0 - 7200.0 * k2 / 2197.0 + 7296.0 * k3 / 2197.0);
38 double k5 = h * f(t + h, y + 439.0 * k1 / 216.0 - 8.0 * k2 + 3680.0 * k3 / 513.0 - 845.0 * k4 / 4104.0);
39 double k6 = h * f(t + h / 2.0, y - 8.0 * k1 / 27.0 + 2.0 * k2 - 3544.0 * k3 / 2565.0 + 1859.0 * k4 / 4104.0 - 11.0 * k5 / 40.0);
40
41 double y4 = y + 25.0 * k1 / 216.0 + 1408.0 * k3 / 2565.0 + 2197.0 * k4 / 4104.0 - k5 / 5.0;
42 double y5 = y + 16.0 * k1 / 135.0 + 6656.0 * k3 / 12825.0 + 28561.0 * k4 / 56430.0 - 9.0 * k5 / 50.0 + 2.0 * k6 / 55.0;
43 return {y4, std::abs(y5 - y4)};
44 }

46 std::vector<std::pair<double, double>> RungeKuttaFehlbergSolver::rkf45(std::function<double(double, double)> f, double t0, double y0,
47 double t_end, double h_initial, double tol)
48 {
49 checks::check_ode_inputs(t0, t_end, h_initial);
50 checks::check_tolerance(tol, 1);
51
52 std::vector<std::pair<double, double>> result;
53 result.push_back({t0, y0});
54
55 double t = t0;
56 double y = y0;
57 double h = std::min(h_initial, t_end - t0);
58 double h_max = t_end - t0;
59 int attempts = 0;
60
61 while (t < t_end)
62 {
63 // Every trial step counts, accepted or not, so a hopeless problem cannot loop forever.
64 if (++attempts > checks::MAX_POINTS)
65 {
66 throw std::runtime_error("RKF45 needed more than 1,000,000 steps: increase tol.");
67 }
68 // A step this small no longer changes t in double precision.
69 if (h < 1e-12 * std::max(1.0, std::abs(t)))
70 {
71 throw std::runtime_error("RKF45 step size became too small (singular problem or tol too strict).");
72 }
73
74 // Do not step past t_end.
75 bool last = (t + h >= t_end);
76 if (last) h = t_end - t;
77
78 FehlbergStep step = fehlberg_step(f, t, y, h);
79 checks::check_finite_result(step.y4);
80 double R = step.error / h; // estimated error per unit time
81
82 if (R <= tol)
83 {
84 t = last ? t_end : t + h; // land exactly on t_end
85 y = step.y4;
86 result.push_back({t, y});
87 }
88
89 // Optimal rescaling factor, limited to [0.1, 4] so h never changes abruptly.
90 double factor = (R > 0.0) ? 0.84 * std::pow(tol / R, 0.25) : 4.0;
91 factor = std::min(4.0, std::max(0.1, factor));
92 h = std::min(h * factor, h_max);
93 }
94
95 return result;
96 }
