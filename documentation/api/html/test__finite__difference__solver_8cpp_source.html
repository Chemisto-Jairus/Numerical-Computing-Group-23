29double max_error(const std::vector<std::pair<double, double>>& grid, const std::function<double(double)>& exact)
30{
31 double worst = 0.0;
32 for (const std::pair<double, double>& point : grid)
33 {
34 worst = std::max(worst, std::abs(point.second - exact(point.first)));
35 }

40void accuracy()
41{
42 numcomp::FiniteDifferenceSolver fd;
43 const double pi = std::acos(-1.0);
44 auto load = [=](double x) { return -pi * pi * std::sin(pi * x); };
45 auto exact_sin = [=](double x) { return std::sin(pi * x); };
46
47 check("Poisson y'' = -pi^2 sin(pi x): n = 100 error below 1e-4", max_error(fd.poisson(load, 0.0, 1.0, 0.0, 0.0, 100), exact_sin) < 1e-4);
48
49 // A quadratic solution is reproduced exactly: the central difference is exact for quadratics.
50 auto quad = fd.poisson([](double) { return 2.0; }, 0.0, 1.0, 0.0, 1.0, 10);
51 check("y'' = 2, y(0) = 0, y(1) = 1 gives y = x^2 exactly", max_error(quad, [](double x) { return x * x; }) < 1e-12);
52
53 // y'' = y (p = 0, q = 1, r = 0) on [0, 1], y(0) = 0, y(1) = sinh(1): exact y = sinh(x).
54 auto zero = [](double) { return 0.0; };
55 auto one = [](double) { return 1.0; };
56 auto grid = fd.linear_bvp(zero, one, zero, 0.0, 1.0, 0.0, std::sinh(1.0), 50);
57 check("y'' = y gives sinh(x) (error below 1e-5)", max_error(grid, [](double x) { return std::sinh(x); }) < 1e-5);
58
59 // Burden & Faires example: y'' = -(2/x) y' + (2/x^2) y + sin(ln x)/x^2 on [1, 2], y(1) = 1, y(2) = 2.
60 double c2 = (8.0 - 12.0 * std::sin(std::log(2.0)) - 4.0 * std::cos(std::log(2.0))) / 70.0;
61 double c1 = 1.1 - c2;
62 auto exact_bf = [=](double x) { return c1 * x + c2 / (x * x) - 0.3 * std::sin(std::log(x)) - 0.1 * std::cos(std::log(x)); };
63 auto bf = fd.linear_bvp([](double x) { return -2.0 / x; }, [](double x) { return 2.0 / (x * x); },
64 [](double x) { return std::sin(std::log(x)) / (x * x); }, 1.0, 2.0, 1.0, 2.0, 10);
65 check("Textbook problem with variable p, q, r: n = 10 error below 1e-4", max_error(bf, exact_bf) < 1e-4);
66
67 double e1 = max_error(fd.poisson(load, 0.0, 1.0, 0.0, 0.0, 20), exact_sin);
68 double e2 = max_error(fd.poisson(load, 0.0, 1.0, 0.0, 0.0, 40), exact_sin);
69 check("Doubling n divides the error by about 4 (second order)", e1 / e2 > 3.8 && e1 / e2 < 4.2);
70}

73void boundaries()
74{
75 numcomp::FiniteDifferenceSolver fd;
76 auto grid = fd.poisson([](double) { return 0.0; }, -1.0, 3.0, 5.0, -3.0, 8);
77 check("n = 8 gives 9 grid points", grid.size() == 9);
78 check("First point is (a, alpha)", grid.front().first == -1.0 && grid.front().second == 5.0);
79 check("Last point is exactly (b, beta)", grid.back().first == 3.0 && grid.back().second == -3.0);
80 check("y'' = 0 gives the straight line between the boundary values", close(grid[4].second, 1.0, 1e-12));
81
82 auto smallest = fd.poisson([](double) { return 2.0; }, 0.0, 2.0, 0.0, 4.0, 2);
83 check("n = 2 (one inner point) gives y(1) = 1 for y = x^2", smallest.size() == 3 && close(smallest[1].second, 1.0, 1e-12));
