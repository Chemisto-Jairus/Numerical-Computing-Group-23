16 std::vector<std::pair<double, double>> FiniteDifferenceSolver::linear_bvp(std::function<double(double)> p, std::function<double(double)> q,
17 std::function<double(double)> r, double a, double b,
18 double alpha, double beta, int n)
19 {
20 // Shared checks: n > 0 and finite a, b. Then the rules specific to this method.
21 checks::check_integration_inputs(a, b, n);
22 if (n < 2)
23 {
24 throw std::invalid_argument("The grid needs at least 2 intervals (one inner point).");
25 }
26 if (n > checks::MAX_POINTS)
27 {
28 throw std::invalid_argument("Number of grid intervals must not be more than 1,000,000.");
29 }
30 if (!(a < b))
31 {
32 throw std::invalid_argument("a must be less than b.");
33 }
34 if (!std::isfinite(alpha) || !std::isfinite(beta))
35 {
36 throw std::invalid_argument("Boundary values alpha and beta must be finite numbers.");
37 }
38
39 double h = (b - a) / n;
40 int m = n - 1; // number of inner grid points, i.e. unknowns
41
42 std::vector<double> lower(m), diag(m), upper(m), rhs(m);
43 for (int k = 0; k < m; ++k)
44 {
45 double x = a + (k + 1) * h; // inner point x_i with i = k + 1
46 double p_i = p(x);
47 double q_i = q(x);
48 double r_i = r(x);
49 checks::check_finite_result(p_i + q_i + r_i);
50
51 lower[k] = -(1.0 + h * p_i / 2.0); // coefficient of y_(i-1)
52 diag[k] = 2.0 + h * h * q_i; // coefficient of y_i
53 upper[k] = -(1.0 - h * p_i / 2.0); // coefficient of y_(i+1)
54 rhs[k] = -h * h * r_i;
55 }
56
57 // Known boundary values move to the right-hand side.
58 rhs[0] -= lower[0] * alpha;
59 rhs[m-1] -= upper[m-1] * beta;
60
61 // Shared Thomas algorithm (also used by CubicSpline). The vectors are not needed afterwards.
62 std::vector<double> inner = linalg::solve_tridiagonal(std::move(lower), std::move(diag),
63 std::move(upper), std::move(rhs));
64
65 // Assemble the full grid, including both boundary points.
66 std::vector<std::pair<double, double>> grid;
67 grid.reserve(n + 1);
68 grid.push_back({a, alpha});
69 for (int k = 0; k < m; ++k)
70 {
71 grid.push_back({a + (k + 1) * h, inner[k]});
72 }
73 grid.push_back({b, beta}); // exactly b, without rounding from a + n h
74 return grid;
75 }
