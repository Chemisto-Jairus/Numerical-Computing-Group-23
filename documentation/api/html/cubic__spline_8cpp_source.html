14 std::vector<double> CubicSpline::second_derivatives(const std::vector<double>& x_vals, const std::vector<double>& y_vals)
15 {
16 checks::check_data_points(x_vals, y_vals);
17
18 int n = static_cast<int>(x_vals.size());
19
20 // Natural boundary conditions: M[0] = M[n-1] = 0.
21 std::vector<double> M(n, 0.0);
22
23 // Two points have no inner point; the spline is the straight line between them.
24 if (n == 2)
25 {
26 return M;
27 }
28
29 // Interval widths h[i] = x[i+1] - x[i].
30 std::vector<double> h(n - 1);
31 for (int i = 0; i < n - 1; ++i)
32 {
33 h[i] = x_vals[i+1] - x_vals[i];
34 }
35
36 // Tridiagonal system for the m = n - 2 inner unknowns M[1] .. M[n-2].
37 int m = n - 2;
38 std::vector<double> lower(m); // sub-diagonal
39 std::vector<double> diag(m); // main diagonal
40 std::vector<double> upper(m); // super-diagonal
41 std::vector<double> rhs(m); // right-hand side
42
43 for (int k = 0; k < m; ++k)
44 {
45 int i = k + 1; // data point that row k belongs to
46 lower[k] = h[i-1];
47 diag[k] = 2.0 * (h[i-1] + h[i]);
48 upper[k] = h[i];
49 rhs[k] = 6.0 * ((y_vals[i+1] - y_vals[i]) / h[i] - (y_vals[i] - y_vals[i-1]) / h[i-1]);
50 }
51
52 // Solve the system with the shared Thomas algorithm (also used by FiniteDifferenceSolver).
53 // The matrix is strictly diagonally dominant, so no pivot can become zero.
54 // std::move hands the vectors over without copying them; they are not used again.
55 std::vector<double> inner = linalg::solve_tridiagonal(std::move(lower), std::move(diag),
56 std::move(upper), std::move(rhs));
57
58 for (int k = 0; k < m; ++k)
59 {
60 M[k+1] = inner[k];
61 }
62
63 return M;
64 }

66 double CubicSpline::interpolate(double x, const std::vector<double>& x_vals, const std::vector<double>& y_vals)
67 {
68 // Also validates the data.
69 std::vector<double> M = second_derivatives(x_vals, y_vals);
70
71 std::size_t i = checks::find_interval(x, x_vals);
72
73 double h = x_vals[i+1] - x_vals[i];
74 double t0 = x - x_vals[i]; // distance from the left data point
75 double t1 = x_vals[i+1] - x; // distance from the right data point
76
77 // Cubic piece on [x_i, x_(i+1)] written in terms of the end values and M.
78 double part1 = M[i] * t1 * t1 * t1 / (6.0 * h);
79 double part2 = M[i+1] * t0 * t0 * t0 / (6.0 * h);
80 double part3 = (y_vals[i] / h - M[i] * h / 6.0) * t1;
81 double part4 = (y_vals[i+1] / h - M[i+1] * h / 6.0) * t0;
82
83 return part1 + part2 + part3 + part4;
84 }
85}
