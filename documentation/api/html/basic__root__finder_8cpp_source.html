16 {
17 checks::check_tolerance(tol, max_iter);
18
19 // Work with a <= b so that (b - a) / 2 is always the true half-width.
20 if (a > b)
21 {
22 std::swap(a, b);
23 }
24
25 // f is evaluated once per point; fa is kept up to date as a moves.
26 double fa = f(a);
27 double fb = f(b);
28
29 // NaN compares false with everything, so it must be rejected explicitly.
30 if (std::isnan(fa) || std::isnan(fb))
31 {
32 throw std::invalid_argument("f(a) or f(b) is not a number (NaN).");
33 }
34
35 // A root sitting exactly on an end point is returned straight away.
36 if (std::abs(fa) < tol) return a;
37 if (std::abs(fb) < tol) return b;
38
39 // Without a sign change the interval is not guaranteed to contain a root.
40 if (fa * fb > 0)
41 {
42 throw std::invalid_argument("Function must have opposite signs at interval endpoints.");
43 }
44
45 for (int iter = 0; iter < max_iter; ++iter)
46 {
47 double c = (a + b) / 2.0;
48 double fc = f(c);
49
50 // Converged: the residual is small or the bracket is narrower than tol.
51 if (std::abs(fc) < tol || (b - a) / 2.0 < tol)
52 {
53 return c;
54 }
55
56 // Keep the half [a, c] or [c, b] in which f still changes sign.
57 if (fc * fa < 0)
58 {
59 b = c;
60 }
61 else
62 {
63 a = c;
64 fa = fc;
65 }
66 }
67 throw std::runtime_error("Bisection method failed to converge.");
68 }

70 double BasicRootFinder::newton_raphson(std::function<double(double)> f, std::function<double(double)> df, double x0, double tol, int max_iter)
71 {
72 checks::check_tolerance(tol, max_iter);
73
74 double x = x0;
75 for (int iter = 0; iter < max_iter; ++iter)
76 {
77 double derivative = df(x);
78
79 // A (nearly) flat tangent never reaches the x-axis, so the step is undefined.
80 if (std::abs(derivative) < 1e-12)
81 {
82 throw std::runtime_error("Derivative too close to zero.");
83 }
84
85 // Move to where the tangent line at x crosses the x-axis.
86 double next_x = x - f(x) / derivative;
87 if (std::abs(next_x - x) < tol)
88 {
89 return next_x;
90 }
91 x = next_x;
92 }
93 throw std::runtime_error("Newton-Raphson failed to converge.");
94 }
