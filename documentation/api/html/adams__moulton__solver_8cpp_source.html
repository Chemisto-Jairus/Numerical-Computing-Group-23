37 static double solve_corrector(const ode::ODEFunction& f, double t_next, double known, double beta_h, double guess)
38 {
39 double y = guess;
40 for (int i = 0; i < MAX_CORRECTOR_ITERATIONS; ++i)
41 {
42 double y_new = known + beta_h * f(t_next, y);
43 checks::check_finite_result(y_new);
44 if (std::abs(y_new - y) <= 1e-12 * (1.0 + std::abs(y_new)))
45 {
46 return y_new;
47 }
48 y = y_new;
49 }
50 throw std::runtime_error("Adams-Moulton corrector did not converge: use a smaller step size h.");

78 static double am4_step(const ode::ODEFunction& f, const std::vector<double>& slopes, double t, double y, double h)
79 {
80 std::size_t n = slopes.size() - 1; // index of the current point
81 double known = y + (h / 24.0) * (19.0 * slopes[n] - 5.0 * slopes[n-1] + slopes[n-2]);
